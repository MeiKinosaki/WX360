cmake_minimum_required(VERSION 3.20)
project(Onix360 VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Keep Windows SDK min/max macros out of all translation units.
if(WIN32)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()

# Statically link the MSVC runtime so the Windows portable build does not
# require a separately installed Visual C++ runtime redistributable.
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

# x86-64 guarantees SSE2. Do not enable AVX/AVX2 or -march=native.

# libmspack regular LZX decoder (LGPL-2.1-or-later).
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(onix360_lzx STATIC
  third_party/libmspack/lzxd.c
  src/compression/lzx_decoder.cpp
)
target_include_directories(onix360_lzx PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}/third_party/libmspack
)
target_include_directories(onix360_lzx PUBLIC
  ${CMAKE_CURRENT_SOURCE_DIR}/include
)
target_compile_definitions(onix360_lzx PRIVATE
  HAVE_INTTYPES_H=1 HAVE_LIMITS_H=1 SIZEOF_OFF_T=8
)

add_library(onix360_aes STATIC third_party/tiny_aes/aes.c)
target_include_directories(onix360_aes PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/tiny_aes)
target_compile_definitions(onix360_aes PRIVATE AES128=1 CBC=1 ECB=1 CTR=0)

add_library(onix360_cpu INTERFACE)
target_include_directories(onix360_cpu INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(onix360_cpu INTERFACE onix360_aes onix360_lzx)
target_sources(onix360_cpu INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src/audio/windows_pcm_output.cpp)
if(WIN32)
  target_link_libraries(onix360_cpu INTERFACE xinput9_1_0 winmm d3d11 dxgi)
endif()

add_executable(onix360 src/main.cpp)
set_target_properties(onix360 PROPERTIES OUTPUT_NAME "Onix360")
target_link_libraries(onix360 PRIVATE onix360_cpu)

# Standalone diagnostic executable for users to verify the emulator core.
add_executable(onix360_test_runner tools/onix360_test_runner.cpp)
set_target_properties(onix360_test_runner PROPERTIES OUTPUT_NAME "Onix360-Test")
target_link_libraries(onix360_test_runner PRIVATE onix360_cpu)

# Bounded diagnostic launcher for user-supplied XEX homebrew images.
add_executable(onix360_xex_probe tools/onix360_xex_probe.cpp)
set_target_properties(onix360_xex_probe PROPERTIES OUTPUT_NAME "Onix360-XEX-Probe")
target_link_libraries(onix360_xex_probe PRIVATE onix360_cpu)
